    definitions: [],
    calls: [call('c1', 'github', 'search'), call('c1', 'github', 'get_file'),

### Step 2: matrixTree.ts 구현

```ts
// packages/extension/src/matrixTree.ts — vscode import 금지 (테스트 가능성)
import type { EngineInput, MatrixRow } from '@mcp-lens/core';
import { buildMatrix, splitByClient } from '@mcp-lens/core';

export interface LensNode {
  kind: 'client' | 'server' | 'tool';
  id: string;
  label: string;
  description: string;
  tooltip?: string;
  /** 'inspect'|'remove' 등 — 래퍼가 아이콘 선택에 사용 */
  action?: MatrixRow['recommendation']['action'];
  children: LensNode[];
}

export function buildTree(input: EngineInput): LensNode[] {
  return splitByClient(input).map(({ client, input: sub }) => {
    const rows = buildMatrix(sub)
      .filter((r) => r.callCount > 0 || r.definitionTokens != null);
    const byServer = new Map<string, MatrixRow[]>();
    for (const r of rows) {
      if (!byServer.has(r.serverName)) byServer.set(r.serverName, []);
      byServer.get(r.serverName)!.push(r);
    }
    const servers: LensNode[] = [...byServer.entries()].map(([server, rs]) => ({
      kind: 'server',
      id: `${client}/${server}`,
      label: server,
      description: `${rs.reduce((s, r) => s + r.callCount, 0)}회`,
      children: rs.map((r): LensNode => ({
        kind: 'tool',
        id: `${client}/${server}/${r.toolName}`,
        label: r.toolName,
        description: `${r.callCount}회 · ${r.responseTokens}tok · ${r.recommendation.action}`,
        tooltip: r.recommendation.reason,
        action: r.recommendation.action,
        children: [],
      })),
    }));
    return {
      kind: 'client' as const, id: client, label: client,
      description: `세션 ${sub.sessions.length}개`, children: servers,
    };
  });
}
```

### Step 3: lensTreeProvider.ts (vscode 래퍼)

```ts
// packages/extension/src/lensTreeProvider.ts
import * as vscode from 'vscode';
import type { LensNode } from './matrixTree.js';

const ACTION_ICON: Record<string, string> = {
  inspect: 'warning', remove: 'trash', 'filter-server': 'filter', keep: 'check',
};

export class LensTreeProvider implements vscode.TreeDataProvider<LensNode> {
  private emitter = new vscode.EventEmitter<void>();
  readonly onDidChangeTreeData = this.emitter.event;
  private roots: LensNode[] = [];

  setRoots(roots: LensNode[]): void { this.roots = roots; this.emitter.fire(); }

  getTreeItem(n: LensNode): vscode.TreeItem {
    const item = new vscode.TreeItem(n.label,
      n.children.length
        ? vscode.TreeItemCollapsibleState.Collapsed
        : vscode.TreeItemCollapsibleState.None);
    item.id = n.id;
    item.description = n.description;
    item.tooltip = n.tooltip;
    if (n.kind === 'client') item.iconPath = new vscode.ThemeIcon('device-desktop');
    else if (n.kind === 'server') item.iconPath = new vscode.ThemeIcon('server');
    else if (n.action) item.iconPath = new vscode.ThemeIcon(ACTION_ICON[n.action] ?? 'circle-outline');
    return item;
  }
  getChildren(n?: LensNode): LensNode[] { return n ? n.children : this.roots; }
}
```

## 2. Task 17 — 배포 패키징 (.vsix) [T1, Task 16 뒤]

**Files:**
- Create: `packages/extension/esbuild.mjs`, `packages/extension/.vscodeignore`, `packages/extension/README.md`, `packages/extension/LICENSE`
- Modify: `packages/extension/package.json` (publisher·scripts·main 경로)

**⚠ 핵심 리스크 (이 태스크가 opus인 이유):** npm 워크스페이스에서 `@mcp-lens/core`는 루트 node_modules에 호이스팅된 심링크라 `vsce package`가 의존성을 .vsix에 담지 못한다. **esbuild로 단일 CJS 번들**이 표준 해법인데, 두 자산이 번들을 깨뜨린다:
1. **sql.js의 `sql-wasm.wasm`** — 런타임에 파일로 로드 → `dist/`에 복사하고 로케이션이 맞는지 실기기 확인
2. **core의 `schema.sql`** — `import.meta.url` 기준 로드 → CJS 번들에서 `import.meta.url`이 깨짐. esbuild banner로 `import.meta.url → pathToFileURL(__filename)` 셰임 주입 + `schema.sql`을 dist에 복사

**Steps:**
1. `esbuild.mjs`: entry `src/extension.ts` → `dist/extension.js` (bundle, platform=node, format=cjs, external: ['vscode']), banner 셰임, 빌드 후 `sql-wasm.wasm`·`schema.sql` 복사
2. package.json: `"publisher": "doohyun"`(가칭 — 마켓플레이스 계정명으로 교체 필요), `"license": "MIT"`, build 스크립트를 esbuild로 교체, `"vscode:prepublish"` 추가
3. `.vscodeignore`: `src/`, `test/`, `tsconfig*`, `esbuild.mjs`, `*.map` 제외
4. `npx @vscode/vsce package --no-dependencies` → `mcp-lens-0.1.0.vsix`
5. **수용 기준 (실기기)**: `code --install-extension mcp-lens-0.1.0.vsix` 후 VS Code 재시작 → F5 없이 상태바·사이드바가 뜨고 백필이 도는지. 실패 시 개발자 도구 콘솔로 디버깅 (wasm/schema 경로 문제가 1순위 용의자)
6. LICENSE는 MIT 전문 + 저작권자 이름은 사용자 확인 필요 → 보고에 명시
7. `git commit -m "build: esbuild bundle + vsce packaging (.vsix)"`
