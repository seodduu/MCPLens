# Claude Code 서버 단위 Remove/Add 구현 계획 — 터미널 병렬 운용 포함

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 트리에서 claude-code 서버를 우클릭 Remove/Add 하면 MCP 서버 자체가 비활성/복원되고, 사용 0회인 설정 서버도 트리에 표시된다.

**Architecture:** `.mcp.json` 서버는 `~/.claude/settings.json`의 `disabledMcpjsonServers` 공식 스위치로, `~/.claude.json` 서버는 스펙을 `~/.mcplens/disabled-servers.json` 스태시로 무손실 이동/복원. 설정 파일들이 유일한 원본. 스펙: `docs/superpowers/specs/2026-07-20-claude-code-server-toggle-design.md`

**Tech Stack:** 기존 그대로 (신규 의존성 없음)

## 전역 제약 (모든 태스크에 적용)

- core에서 `vscode` import 금지. 경로는 `os.homedir()` + `path.join()`.
- **읽기는 관대**(깨진 파일은 건너뜀), **쓰기는 엄격**(파싱 실패 시 throw + 미변경).
- 쓰기 전 `.mcplens.bak` 백업 + tmp/rename 원자적 쓰기 (`fileUtils.ts` 경유).
- 스펙 유실 금지 순서: disable은 **스태시 먼저 쓰고** `~/.claude.json`에서 제거, enable은 **`~/.claude.json`에 먼저 복원하고** 스태시에서 제거.
- 유닛테스트는 temp 디렉토리·주입 데이터만. 실기기 `~/.claude*` 접근은 Wave 2 E2E 전용.
- 커밋 메시지는 conventional commits.

```ts
// packages/core/src/permissions/fileUtils.ts — vscode import 금지
// permissions 모듈 공용 JSON 파일 IO. 원본 보호: 쓰기 전 .mcplens.bak 백업 + tmp/rename.
import * as fs from 'node:fs';
import * as path from 'node:path';

/** JSON 파일 읽기. 파일 없으면 null. 파싱 실패는 throw — 관대/엄격은 호출부가 결정한다. */
export function readJsonFile(filePath: string): unknown | null {
  if (!fs.existsSync(filePath)) return null;
  return JSON.parse(fs.readFileSync(filePath, 'utf8'));
}

/** 원본 보호 쓰기: 기존 파일을 .mcplens.bak으로 백업 후 tmp에 쓰고 rename. */
export function writeJsonFile(filePath: string, value: unknown): void {
  fs.mkdirSync(path.dirname(filePath), { recursive: true });
  if (fs.existsSync(filePath)) fs.copyFileSync(filePath, `${filePath}.mcplens.bak`);
  const tmp = `${filePath}.mcplens.tmp`;
  fs.writeFileSync(tmp, `${JSON.stringify(value, null, 2)}\n`);
  fs.renameSync(tmp, filePath);
}
```

```ts
// packages/core/test/serverToggle.test.ts
import { describe, it, expect, beforeEach, afterEach } from 'vitest';
import * as fs from 'node:fs';
import * as os from 'node:os';
import * as path from 'node:path';
import { listManagedServers, disableServer, enableServer,
  type ClaudeConfigPaths } from '../src/permissions/claudeServers.js';

const FIXTURES = path.join(__dirname, '..', '..', '..', 'fixtures', 'claude-settings');

let dir: string;
let paths: ClaudeConfigPaths;
const put = (fixture: string, dest: string) =>
  fs.copyFileSync(path.join(FIXTURES, fixture), dest);
const readJson = (p: string) => JSON.parse(fs.readFileSync(p, 'utf8'));

beforeEach(() => {
  dir = fs.mkdtempSync(path.join(os.tmpdir(), 'mcplens-srv-'));
  paths = {
    settingsPath: path.join(dir, 'settings.json'),
    claudeJsonPath: path.join(dir, 'claude.json'),
    stashPath: path.join(dir, 'disabled-servers.json'),
    mcpJsonPaths: [path.join(dir, 'mcp-project.json')],
  };
});
afterEach(() => fs.rmSync(dir, { recursive: true, force: true }));

describe('listManagedServers', () => {
  it('아무 파일도 없으면 빈 배열', () => {
    expect(listManagedServers(paths)).toEqual([]);
  });
  it('유저 전역(top-level·projects)과 .mcp.json 서버를 수집한다', () => {
    put('claude-user.json', paths.claudeJsonPath);
    put('mcp-project.json', paths.mcpJsonPaths[0]);
    const byName = new Map(listManagedServers(paths).map((m) => [m.name, m]));
    expect(byName.get('lenstest')).toEqual({ name: 'lenstest', enabled: true, origins: ['user'] });
    expect(byName.get('projserver')).toEqual({ name: 'projserver', enabled: true, origins: ['user'] });
    expect(byName.get('shared-db')).toEqual({ name: 'shared-db', enabled: true, origins: ['project'] });
  });
  it('스태시에 있는 서버는 enabled=false로 표시된다 (계속 보임 — 복원용)', () => {
    put('stash-basic.json', paths.stashPath);
    const byName = new Map(listManagedServers(paths).map((m) => [m.name, m]));
    expect(byName.get('oldserver')).toEqual({ name: 'oldserver', enabled: false, origins: ['user'] });
  });
  it('disabledMcpjsonServers에 있는 .mcp.json 서버는 enabled=false', () => {
    put('mcp-project.json', paths.mcpJsonPaths[0]);
    fs.writeFileSync(paths.settingsPath, JSON.stringify({ disabledMcpjsonServers: ['shared-db'] }));
    const byName = new Map(listManagedServers(paths).map((m) => [m.name, m]));
    expect(byName.get('shared-db')!.enabled).toBe(false);
  });
  it('깨진 파일은 건너뛰고 나머지는 수집한다 (읽기 관대)', () => {
    put('broken.json', paths.claudeJsonPath);
    put('mcp-project.json', paths.mcpJsonPaths[0]);
    const names = listManagedServers(paths).map((m) => m.name);
    expect(names).toEqual(['shared-db']);
  });
});

describe('disableServer — 유저 전역(~/.claude.json)', () => {
  beforeEach(() => put('claude-user.json', paths.claudeJsonPath));

  it('top-level 서버를 스태시로 이동하고 claude.json에서 제거한다', () => {
    disableServer(paths, 'lenstest');
    const cj = readJson(paths.claudeJsonPath);
    expect(cj.mcpServers.lenstest).toBeUndefined();
    expect(cj.oauthAccount.email).toBe('user@example.com'); // 다른 키 보존
    const stash = readJson(paths.stashPath);
    expect(stash.lenstest.origin).toEqual({ kind: 'user' });
    expect(stash.lenstest.spec).toEqual({ command: 'node', args: ['server.js'], env: { PORT: '3100' } });
  });
  it('projects.<path> 서버는 user-project origin으로 스태시된다', () => {
    disableServer(paths, 'projserver');
    const cj = readJson(paths.claudeJsonPath);
    expect(cj.projects['/Users/x/proj'].mcpServers.projserver).toBeUndefined();
    expect(cj.projects['/Users/x/proj'].allowedTools).toEqual([]); // 형제 키 보존
    const stash = readJson(paths.stashPath);
    expect(stash.projserver.origin).toEqual({ kind: 'user-project', projectPath: '/Users/x/proj' });
  });
  it('쓰기 전 claude.json 백업을 남긴다', () => {
    const before = fs.readFileSync(paths.claudeJsonPath, 'utf8');
    disableServer(paths, 'lenstest');
    expect(fs.readFileSync(`${paths.claudeJsonPath}.mcplens.bak`, 'utf8')).toBe(before);
  });
  it('claude.json이 깨져 있으면 throw + 미변경 (쓰기 엄격)', () => {
    put('broken.json', paths.claudeJsonPath);
    const before = fs.readFileSync(paths.claudeJsonPath, 'utf8');
    expect(() => disableServer(paths, 'lenstest')).toThrow();
    expect(fs.readFileSync(paths.claudeJsonPath, 'utf8')).toBe(before);
    expect(fs.existsSync(paths.stashPath)).toBe(false);
  });
  it('이미 스태시된 서버는 no-op (멱등)', () => {
    disableServer(paths, 'lenstest');
    const stashBefore = fs.readFileSync(paths.stashPath, 'utf8');
    disableServer(paths, 'lenstest');
    expect(fs.readFileSync(paths.stashPath, 'utf8')).toBe(stashBefore);
  });
});

describe('disableServer — 프로젝트 .mcp.json', () => {
  it('disabledMcpjsonServers에 이름을 추가한다 (.mcp.json은 건드리지 않음)', () => {
    put('mcp-project.json', paths.mcpJsonPaths[0]);
    const mcpBefore = fs.readFileSync(paths.mcpJsonPaths[0], 'utf8');
    disableServer(paths, 'shared-db');
    expect(readJson(paths.settingsPath).disabledMcpjsonServers).toEqual(['shared-db']);
    expect(fs.readFileSync(paths.mcpJsonPaths[0], 'utf8')).toBe(mcpBefore);
    expect(fs.existsSync(paths.stashPath)).toBe(false); // 스태시 미사용
  });
  it('settings의 기존 키를 보존한다', () => {
    put('mcp-project.json', paths.mcpJsonPaths[0]);
    fs.writeFileSync(paths.settingsPath, JSON.stringify({ model: 'opus' }));
    disableServer(paths, 'shared-db');
    const s = readJson(paths.settingsPath);
    expect(s.model).toBe('opus');
    expect(s.disabledMcpjsonServers).toEqual(['shared-db']);
  });
});

describe('enableServer', () => {
  it('스태시의 top-level 서버를 원위치로 무손실 복원한다', () => {
    put('claude-user.json', paths.claudeJsonPath);
    const specBefore = readJson(paths.claudeJsonPath).mcpServers.lenstest;
    disableServer(paths, 'lenstest');
    enableServer(paths, 'lenstest');
    expect(readJson(paths.claudeJsonPath).mcpServers.lenstest).toEqual(specBefore);
    expect(readJson(paths.stashPath)).toEqual({}); // 스태시 비워짐
  });
  it('user-project 서버를 원래 프로젝트 경로로 복원한다', () => {
    put('claude-user.json', paths.claudeJsonPath);
    disableServer(paths, 'projserver');
```

```ts
/** 읽기 관대 버전: 깨진 파일은 null로 취급. */
function readLenient(filePath: string): unknown | null {
  try { return readJsonFile(filePath); } catch { return null; }
}

function serverNames(obj: unknown): string[] {
  const m = (obj as { mcpServers?: unknown } | null)?.mcpServers;
  return m && typeof m === 'object' ? Object.keys(m) : [];
}

function disabledMcpjson(settings: unknown): string[] {
  const list = (settings as { disabledMcpjsonServers?: unknown } | null)?.disabledMcpjsonServers;
  return Array.isArray(list) ? list.filter((n): n is string => typeof n === 'string') : [];
}

export function listManagedServers(paths: ClaudeConfigPaths): ManagedServer[] {
  const disabledJson = new Set(disabledMcpjson(readLenient(paths.settingsPath)));
  const stash = (readLenient(paths.stashPath) ?? {}) as Record<string, StashedServer>;

  const origins = new Map<string, Set<'user' | 'project'>>();
  const add = (name: string, origin: 'user' | 'project') => {
    if (!origins.has(name)) origins.set(name, new Set());
    origins.get(name)!.add(origin);
  };

  const cj = readLenient(paths.claudeJsonPath) as
    { mcpServers?: unknown; projects?: Record<string, unknown> } | null;
  for (const name of serverNames(cj)) add(name, 'user');
  for (const proj of Object.values(cj?.projects ?? {})) {
    for (const name of serverNames(proj)) add(name, 'user');
  }
  for (const p of paths.mcpJsonPaths) {
    for (const name of serverNames(readLenient(p))) add(name, 'project');
  }
  for (const name of Object.keys(stash)) add(name, 'user'); // 스태시는 항상 유저 전역 출신

  return [...origins.entries()].map(([name, o]) => ({
    name,
    enabled: !disabledJson.has(name) && !(name in stash),
    origins: [...o],
  }));
}

export function disableServer(paths: ClaudeConfigPaths, name: string): void {
  // 1) 유저 전역: ~/.claude.json → 스태시 이동. 쓰기 대상이므로 엄격 읽기(깨졌으면 throw).
  const cj = readJsonFile(paths.claudeJsonPath) as
    { mcpServers?: Record<string, unknown>; projects?: Record<string, { mcpServers?: Record<string, unknown> }> } | null;
  let origin: StashOrigin | null = null;
  let spec: unknown;
  if (cj?.mcpServers && name in cj.mcpServers) {
    origin = { kind: 'user' };
    spec = cj.mcpServers[name];
    delete cj.mcpServers[name];
  } else if (cj?.projects) {
    for (const [projectPath, proj] of Object.entries(cj.projects)) {
      if (proj?.mcpServers && name in proj.mcpServers) {
        origin = { kind: 'user-project', projectPath };
        spec = proj.mcpServers[name];
        delete proj.mcpServers[name];
        break;
      }
    }
  }
  if (origin && cj) {
    const stash = (readJsonFile(paths.stashPath) ?? {}) as Record<string, StashedServer>;
    if (!(name in stash)) {
      stash[name] = { origin, spec, disabledAt: new Date().toISOString() };
      writeJsonFile(paths.stashPath, stash);      // 스펙 유실 방지: 스태시 먼저
      writeJsonFile(paths.claudeJsonPath, cj);
    }
  }

  // 2) 프로젝트 .mcp.json: settings의 disabledMcpjsonServers에 추가 (.mcp.json은 읽기 전용).
  const inMcpJson = paths.mcpJsonPaths.some((p) => serverNames(readLenient(p)).includes(name));
  if (inMcpJson) {
    const settings = (readJsonFile(paths.settingsPath) ?? {}) as Record<string, unknown>;
    const list = disabledMcpjson(settings);
    if (!list.includes(name)) {
      settings.disabledMcpjsonServers = [...list, name];
      writeJsonFile(paths.settingsPath, settings);
    }
  }
}

export function enableServer(paths: ClaudeConfigPaths, name: string): void {
  // 1) 스태시 → ~/.claude.json 원위치 복원 (복원 먼저, 스태시 제거는 나중 — 스펙 유실 방지).
  const stash = (readJsonFile(paths.stashPath) ?? {}) as Record<string, StashedServer>;
  const entry = stash[name];
  if (entry) {
    const cj = (readJsonFile(paths.claudeJsonPath) ?? {}) as
      { mcpServers?: Record<string, unknown>; projects?: Record<string, { mcpServers?: Record<string, unknown> }> };
    if (entry.origin.kind === 'user') {
      cj.mcpServers = { ...cj.mcpServers, [name]: entry.spec };
    } else {
      const pp = entry.origin.projectPath;
      cj.projects = { ...cj.projects };
      cj.projects[pp] = { ...cj.projects[pp], mcpServers: { ...cj.projects[pp]?.mcpServers, [name]: entry.spec } };
    }
    writeJsonFile(paths.claudeJsonPath, cj);
    delete stash[name];
    writeJsonFile(paths.stashPath, stash);
  }

  // 2) disabledMcpjsonServers에서 제거.
  const settings = readJsonFile(paths.settingsPath) as Record<string, unknown> | null;
  const list = disabledMcpjson(settings);
  if (settings && list.includes(name)) {
    settings.disabledMcpjsonServers = list.filter((n) => n !== name);
    writeJsonFile(paths.settingsPath, settings);
  }
}
```

```ts
// packages/extension/src/matrixTree.ts — vscode import 금지 (테스트 가능성)
import type { EngineInput, ManagedServer, MatrixRow } from '@mcp-lens/core';
import { buildMatrix, mcpDenyRule, splitByClient, visibleMatrixRows } from '@mcp-lens/core';

export interface LensNode {
  kind: 'client' | 'server' | 'tool';
  id: string;
  label: string;
  description: string;
  tooltip?: string;
  /** 'inspect'|'remove' 등 — 래퍼가 아이콘 선택에 사용 */
  action?: MatrixRow['recommendation']['action'];
  /** deny/서버 토글용 — 서버·툴 노드에 채움 */
  client?: string;
  serverName?: string;
  toolName?: string;
  /** claude-code 하위에만 존재. false = 차단됨(✕). undefined = 토글 불가 */
  enabled?: boolean;
  children: LensNode[];
}

/** 성공은 기본값 — 실패가 있을 때만 괄호로 표기 */
function fmtCalls(calls: number, errors: number): string {
  return errors > 0 ? `${calls}회(실패 ${errors})` : `${calls}회`;
}

/** 사용 이력 없는 managed 서버 → 0회 서버 노드 (스펙 1-2: 설정된 서버는 항상 표시) */
function zeroUsageServerNodes(managed: ManagedServer[], usedNames: Set<string>): LensNode[] {
  return managed.filter((s) => !usedNames.has(s.name)).map((s): LensNode => ({
    kind: 'server',
    id: `claude-code/${s.name}`,
    label: s.name,
    description: s.enabled ? '0회 · 0tok' : '0회 · 0tok · 차단됨',
    client: 'claude-code', serverName: s.name, enabled: s.enabled,
    children: [],
  }));
}

export function buildTree(input: EngineInput, deniedRules: Set<string> = new Set(),
  managedServers: ManagedServer[] = []): LensNode[] {
  const managed = new Map(managedServers.map((s) => [s.name, s]));
  const roots = splitByClient(input).map(({ client, input: sub }) => {
    // 모드 인지 표시: 지연 로딩 클라이언트에서는 실비용 0인 미사용 툴을 숨긴다
    const rows = visibleMatrixRows(buildMatrix(sub), sub.sessions);
    const byServer = new Map<string, MatrixRow[]>();
    for (const r of rows) {
      if (!byServer.has(r.serverName)) byServer.set(r.serverName, []);
      byServer.get(r.serverName)!.push(r);
    }
    const servers: LensNode[] = [...byServer.entries()].map(([server, rs]) => {
      const serverEnabled = client === 'claude-code' ? managed.get(server)?.enabled : undefined;
      const desc = `${fmtCalls(
        rs.reduce((s, r) => s + r.callCount, 0),
        rs.reduce((s, r) => s + r.reliability.errorCount, 0),
      )} · ${rs.reduce((s, r) => s + r.responseTokens, 0)}tok`;
      return {
        kind: 'server' as const,
        id: `${client}/${server}`,
        label: server,
        description: serverEnabled === false ? `${desc} · 차단됨` : desc,
        client, serverName: server, enabled: serverEnabled,
        children: rs.map((r): LensNode => {
          // 서버가 꺼져 있으면 툴 단위 토글은 무의미 — 메뉴 숨김(enabled 미부여)
          const enabled = client === 'claude-code' && serverEnabled !== false
            ? !deniedRules.has(mcpDenyRule(server, r.toolName)) : undefined;
          const toolDesc = `${fmtCalls(r.callCount, r.reliability.errorCount)} · ${r.responseTokens}tok · ${r.recommendation.action}`;
          return {
            kind: 'tool',
            id: `${client}/${server}/${r.toolName}`,
            label: r.toolName,
            description: enabled === false ? `${toolDesc} · 차단됨` : toolDesc,
            tooltip: r.recommendation.reason,
            action: r.recommendation.action,
            client, serverName: server, toolName: r.toolName,
            enabled,
            children: [],
          };
        }),
      };
    });
    if (client === 'claude-code') {
      servers.push(...zeroUsageServerNodes(managedServers, new Set(byServer.keys())));
    }
    const totalCalls = rows.reduce((s, r) => s + r.callCount, 0);
    const totalErrors = rows.reduce((s, r) => s + r.reliability.errorCount, 0);
    const totalTokens = rows.reduce((s, r) => s + r.responseTokens, 0);
    return {
      kind: 'client' as const, id: client, label: client,
      description: `세션 ${sub.sessions.length}개 · 총 ${fmtCalls(totalCalls, totalErrors)} · ${totalTokens}tok`,
      children: servers,
    };
  });
  // 기간 내 claude-code 세션이 없어도 설정된 서버는 표시한다 (스펙 1-2)
  if (managedServers.length > 0 && !roots.some((r) => r.id === 'claude-code')) {
    roots.push({
      kind: 'client', id: 'claude-code', label: 'claude-code',
      description: '세션 0개 · 총 0회 · 0tok',
      children: zeroUsageServerNodes(managedServers, new Set()),
    });
  }
  return roots;
}
```
