# Claude Code 툴 Deny 토글 구현 계획 — 터미널 병렬 운용 포함

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 매트릭스 트리에서 claude-code 툴을 우클릭 Remove/Add 하면 `~/.claude/settings.json`의 `permissions.deny`가 토글되어 새 세션부터 해당 툴 호출이 차단/복원된다.

**Architecture:** settings.json이 유일한 원본(source of truth). core의 순수 모듈(`permissions/claudeSettings.ts`)이 읽기/쓰기를 담당하고, extension은 우클릭 메뉴·아이콘·파일 워처만 얹는 얇은 표면. 스펙: `docs/superpowers/specs/2026-07-20-claude-code-tool-deny-toggle-design.md`

**Tech Stack:** 기존 그대로 — TypeScript(strict), vitest, node:fs (신규 의존성 없음)

## 전역 제약 (모든 태스크에 적용)

- core에서 `vscode` import 금지.
- 경로는 `os.homedir()` + `path.join()` 조립. `~` 하드코딩 금지.
- settings.json **파싱 실패 시 절대 덮어쓰지 않는다** — throw 전파, 쓰기 없음.
- 쓰기 전 `settings.json.mcplens.bak` 백업, tmp+rename 원자적 쓰기.
- 유닛테스트는 temp 디렉토리·주입 데이터만 사용. 실기기 `~/.claude` 접근은 Wave 2 E2E 전용.
- 커밋 메시지는 conventional commits.

```ts
// packages/core/src/permissions/claudeSettings.ts — vscode import 금지
// ~/.claude/settings.json 의 permissions.deny 를 읽고 쓰는 순수 모듈.
// 원본 보호 계약: 파싱 실패 시 절대 쓰지 않는다. 쓰기 전 .mcplens.bak 백업 + tmp/rename.
import * as fs from 'node:fs';
import * as os from 'node:os';
import * as path from 'node:path';

export function claudeSettingsPath(): string {
  return path.join(os.homedir(), '.claude', 'settings.json');
}

/** deny 규칙 문자열 조립. 파싱(쪼개기)은 어디서도 하지 않는다 — 항상 조립해서 대조. */
export function mcpDenyRule(server: string, tool: string): string {
  return `mcp__${server}__${tool}`;
}

interface ClaudeSettings {
  permissions?: { deny?: unknown; [k: string]: unknown };
  [k: string]: unknown;
}

function readSettings(settingsPath: string): ClaudeSettings {
  if (!fs.existsSync(settingsPath)) return {};
  return JSON.parse(fs.readFileSync(settingsPath, 'utf8')) as ClaudeSettings;
}

function denyList(settings: ClaudeSettings): string[] {
  const deny = settings.permissions?.deny;
  return Array.isArray(deny) ? deny.filter((r): r is string => typeof r === 'string') : [];
}

export function readDeniedMcpTools(settingsPath: string): Set<string> {
  return new Set(denyList(readSettings(settingsPath)).filter((r) => r.startsWith('mcp__')));
}

function writeSettings(settingsPath: string, settings: ClaudeSettings): void {
  fs.mkdirSync(path.dirname(settingsPath), { recursive: true });
  if (fs.existsSync(settingsPath)) fs.copyFileSync(settingsPath, `${settingsPath}.mcplens.bak`);
  const tmp = `${settingsPath}.mcplens.tmp`;
  fs.writeFileSync(tmp, `${JSON.stringify(settings, null, 2)}\n`);
  fs.renameSync(tmp, settingsPath);
}

export function denyMcpTool(settingsPath: string, server: string, tool: string): void {
  const settings = readSettings(settingsPath); // 깨진 JSON이면 여기서 throw — 파일 미변경
  const rule = mcpDenyRule(server, tool);
  const deny = denyList(settings);
  if (deny.includes(rule)) return;
  settings.permissions = { ...settings.permissions, deny: [...deny, rule] };
  writeSettings(settingsPath, settings);
}

export function allowMcpTool(settingsPath: string, server: string, tool: string): void {
  const settings = readSettings(settingsPath);
  const rule = mcpDenyRule(server, tool);
  const deny = denyList(settings);
  if (!deny.includes(rule)) return;
  settings.permissions = { ...settings.permissions, deny: deny.filter((r) => r !== rule) };
  writeSettings(settingsPath, settings);
}
```

  const toggleTool = (node: LensNode, action: 'deny' | 'allow') => {
    if (!node?.serverName || !node.toolName) return;
    try {
      if (action === 'deny') {
        denyMcpTool(claudeSettingsPath(), node.serverName, node.toolName);
        void vscode.window.showInformationMessage(
          `${node.serverName}/${node.toolName} 차단됨 — 새 Claude Code 세션부터 적용됩니다.`);
      } else {
        allowMcpTool(claudeSettingsPath(), node.serverName, node.toolName);
      }
    } catch (e) {
      void vscode.window.showErrorMessage(
        `settings.json을 수정하지 못했습니다 (파일은 변경되지 않음): ${e instanceof Error ? e.message : String(e)}`);
    }
    refresh();
  };
  ctx.subscriptions.push(
    vscode.commands.registerCommand('mcp-lens.removeTool', (n: LensNode) => toggleTool(n, 'deny')),
    vscode.commands.registerCommand('mcp-lens.addTool', (n: LensNode) => toggleTool(n, 'allow')));

  const settingsDir = vscode.Uri.file(path.dirname(claudeSettingsPath()));
  const settingsWatcher = vscode.workspace.createFileSystemWatcher(
    new vscode.RelativePattern(settingsDir, 'settings.json'));
  settingsWatcher.onDidChange(refresh);
  settingsWatcher.onDidCreate(refresh);
  settingsWatcher.onDidDelete(refresh);
